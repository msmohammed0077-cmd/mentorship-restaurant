class AddCartItemController {
    constructor( useCase: AddCartItemUseCase) {}

    @AuthDecorators
    @ApiDecorators
    handle(command: AddCartItemCommand) {
        return this.useCase.execute(command);
    }
}


class AddCartItemUseCase {

    constructor(restaurantService: RestaurantService, cartService: CartService, repository: AddCartItemRepository) {}

    execute(command: AddCartItemCommand, userUuid) {

        // error if anything fails
        restaurantService.isRestaurantOpen(command.restaurantId)
        restaurantService.isItemAvailable(command.itemId)

        // error if a cart already exists => prompt to clear cart first
        cartService.createCartForRestaurantIfNotExists()
        repository.addCartItem(command, userUuid)
        cartService.recalculateTotals()

        return

    }
}
